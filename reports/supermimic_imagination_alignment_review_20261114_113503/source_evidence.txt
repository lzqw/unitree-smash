38: class SuperMimicWorldModel(nn.Module):
39:     """Predict task transitions while preserving the native history contract.
40: 
41:     The native 794D observation contains eleven 69D frames.  Treating all
42:     759 history values as independently predicted physical state lets the
43:     imagination model invent inconsistent histories.  For the full-racket
44:     contract, this model predicts only the next frame and the non-history
45:     tail; the history shift itself is deterministic and identical to
46:     ``PolicyHistory.observe``.  The public forward interface remains
47:     ``(delta, reward, terminal_logit)`` so the existing learner can use the
48:     structured transition without an algorithm rewrite.
49:     """
50:     def __init__(self, obs_dim: int, action_dim: int = 18, *, structured_history: bool = True,
51:                  event_dim: int = 0):
52:         super().__init__()
53:         self.obs_dim = obs_dim
54:         self.action_dim = action_dim
55:         self.event_dim = int(event_dim)
56:         if self.event_dim < 0:
57:             raise ValueError("event_dim must be non-negative")
58:         self.structured_history = bool(structured_history and obs_dim == 794)
59:         self.history_length = 11
60:         self.frame_dim = 69
61:         self.history_dim = self.history_length * self.frame_dim
62:         self.tail_dim = obs_dim - self.history_dim
63:         output_dim = (self.frame_dim + self.tail_dim + 2 + self.event_dim
64:                       if self.structured_history else obs_dim + 2 + self.event_dim)
65:         self.body = MLP(obs_dim + action_dim + self.event_dim, (1024, 1024, 1024), output_dim)
66: 
67:     def forward(self, obs: torch.Tensor, action: torch.Tensor, event_state=None):
68:         if self.event_dim:
69:             if event_state is None or event_state.shape[-1] != self.event_dim:
70:                 raise ValueError("event_state is required for an event-aware world model")
71:             model_input = torch.cat((obs, action, event_state), dim=-1)
72:         else:
73:             model_input = torch.cat((obs, action), dim=-1)
74:         out = self.body(model_input)
75:         if self.structured_history:
76:             frame_delta = out[..., :self.frame_dim]
77:             tail_delta = out[..., self.frame_dim:self.frame_dim + self.tail_dim]
78:             history = obs[..., :self.history_dim].reshape(
79:                 *obs.shape[:-1], self.history_length, self.frame_dim)
